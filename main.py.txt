import tkinter as tk
from tkinter import ttk, messagebox
import json
import os
from datetime import datetime

class ExpenseTracker:
    def __init__(self, root):
        self.root = root
        self.root.title("Expense Tracker")
        self.expenses = []

        self.load_data()
        self.create_widgets()

    def load_data(self):
        if os.path.exists("expenses_data.json"):
            try:
                with open("expenses_data.json", "r", encoding="utf-8") as f:
                    self.expenses = json.load(f)
            except Exception as e:
                messagebox.showerror("Ошибка", f"Ошибка загрузки данных: {e}")

    def create_widgets(self):
        # Форма ввода
        input_frame = tk.Frame(self.root)
        input_frame.pack(pady=10, padx=10, fill="x")

        tk.Label(input_frame, text="Сумма:").grid(row=0, column=0, sticky="w")
        self.amount_entry = tk.Entry(input_frame)
        self.amount_entry.grid(row=0, column=1, padx=5)

        tk.Label(input_frame, text="Категория:").grid(row=1, column=0, sticky="w")
        self.category_var = tk.StringVar()
        categories = ["Еда", "Транспорт", "Развлечения", "Жильё", "Одежда", "Другое"]
        self.category_combo = ttk.Combobox(input_frame, textvariable=self.category_var, values=categories)
        self.category_combo.grid(row=1, column=1, padx=5)

        tk.Label(input_frame, text="Дата (ГГГГ-ММ-ДД):").grid(row=2, column=0, sticky="w")
        self.date_entry = tk.Entry(input_frame)
        self.date_entry.insert(0, datetime.now().strftime("%Y-%m-%d"))
        self.date_entry.grid(row=2, column=1, padx=5)

        tk.Button(input_frame, text="Добавить расход", command=self.add_expense).grid(row=3, column=0, columnspan=2, pady=10)

        # Фильтры
        filter_frame = tk.Frame(self.root)
        filter_frame.pack(pady=5, padx=10, fill="x")

        tk.Label(filter_frame, text="Фильтр по категории:").grid(row=0, column=0)
        self.filter_category_var = tk.StringVar()
        self.filter_category = ttk.Combobox(filter_frame, textvariable=self.filter_category_var,
                                    values=["Все"] + categories, state="readonly")
        self.filter_category.grid(row=0, column=1, padx=5)
        self.filter_category.set("Все")

        tk.Label(filter_frame, text="Период с (ГГГГ-ММ-ДД):").grid(row=1, column=0)
        self.start_date_entry = tk.Entry(filter_frame)
        self.start_date_entry.insert(0, (datetime.now().replace(day=1)).strftime("%Y-%m-%d"))
        self.start_date_entry.grid(row=1, column=1, padx=5)

        tk.Label(filter_frame, text="по (ГГГГ-ММ-ДД):").grid(row=2, column=0)
        self.end_date_entry = tk.Entry(filter_frame)
        self.end_date_entry.insert(0, datetime.now().strftime("%Y-%m-%d"))
        self.end_date_entry.grid(row=2, column=1, padx=5)

        tk.Button(filter_frame, text="Применить фильтры", command=self.apply_filters).grid(row=3, column=0, columnspan=2, pady=5)

        # Таблица расходов
        columns = ("ID", "Сумма", "Категория", "Дата")
        self.tree = ttk.Treeview(self.root, columns=columns, show="headings", height=10)
        for col in columns:
            self.tree.heading(col, text=col)
            self.tree.column(col, width=100)
        self.tree.pack(pady=10, padx=10, fill="both", expand=True)

        # Подсчёт суммы
        self.total_label = tk.Label(self.root, text="Общая сумма за период: 0 руб.", font=("Arial", 10, "bold"))
        self.total_label.pack(pady=5)

        # Кнопки управления
        btn_frame = tk.Frame(self.root)
        btn_frame.pack(pady=5)
        tk.Button(btn_frame, text="Сохранить данные", command=self.save_data).pack(side="left", padx=5)
        tk.Button(btn_frame, text="Обновить таблицу", command=self.refresh_table).pack(side="left", padx=5)

        self.refresh_table()

    def add_expense(self):
        try:
            amount = float(self.amount_entry.get())
            if amount <= 0:
                raise ValueError("Сумма должна быть положительным числом")

            category = self.category_var.get()
            if not category:
                raise ValueError("Выберите категорию")

            date_str = self.date_entry.get()
            try:
                date = datetime.strptime(date_str, "%Y-%m-%d")
            except ValueError:
                raise ValueError("Неверный формат даты. Используйте ГГГГ-ММ-ДД")

            expense = {
                "id": len(self.expenses) + 1,
                "amount": amount,
                "category": category,
                "date": date_str
            }
            self.expenses.append(expense)

            # Очистка полей ввода
            self.amount_entry.delete(0, tk.END)
            self.category_combo.set("")
            self.date_entry.delete(0, tk.END)
            self.date_entry.insert(0, datetime.now().strftime("%Y-%m-%d"))

            self.refresh_table()
            messagebox.showinfo("Успех", "Расход успешно добавлен!")
        except ValueError as e:
            messagebox.showerror("Ошибка", str(e))


    def apply_filters(self):
        filtered = self.expenses

        # Фильтр по категории
        category_filter = self.filter_category_var.get()
        if category_filter != "Все":
            filtered = [e for e in filtered if e["category"] == category_filter]

        # Фильтр по дате
        try:
            start_date = datetime.strptime(self.start_date_entry.get(), "%Y-%m-%d")
            end_date = datetime.strptime(self.end_date_entry.get(), "%Y-%m-%d")
            filtered = [
                e for e in filtered
                if start_date <= datetime.strptime(e["date"], "%Y-%m-%d") <= end_date
            ]
        except ValueError:
            messagebox.showerror("Ошибка", "Неверный формат даты в фильтрах")
            return

        self.refresh_table(filtered)
        total = sum(e["amount"] for e in filtered)
        self.total_label.config(text=f"Общая сумма за период: {total:.2f} руб.")

    def refresh_table(self, data=None):
        # Очищаем таблицу
        for item in self.tree.get_children():
            self.tree.delete(item)

        # Если данные не переданы, используем все расходы
        if data is None:
            data = self.expenses

        # Заполняем таблицу данными
        for expense in data:
            self.tree.insert("", "end", values=(
                expense["id"],
                f"{expense['amount']:.2f}",
                expense["category"],
                expense["date"]
            ))

    def save_data(self):
